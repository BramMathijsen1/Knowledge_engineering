03. Fusion genres

Which fusion genres join two genres that had no link before?

SELECT ?fusion ?parentA ?parentB ?alreadyLinked
WHERE {
  ?genre riff:isFusionOf ?a , ?b ;
         rdfs:label ?fusion .
  ?a rdfs:label ?parentA .
  ?b rdfs:label ?parentB .

  BIND(EXISTS { ?a riff:hasStylisticOrigin+ ?b } || EXISTS { ?b riff:hasStylisticOrigin+ ?a } AS ?alreadyLinked)

09. Build the genre routes

If we add up artist influence by genre, which routes do we get and how do they compare with the family tree?

CONSTRUCT {
  ?route a riff:GenreRoute ;
         riff:routeFrom ?from ;
         riff:routeTo ?to ;
         riff:routeWeight ?weight ;
         riff:routeStatus ?status .
}
WHERE {
  {
    SELECT ?from ?to (COUNT(*) AS ?weight)
    WHERE {
      ?follower mm:isInfluencedBy ?artist .
      ?artist riff:hasPrimaryGenre ?from .
      ?follower riff:hasPrimaryGenre ?to .
      FILTER(?from != ?to)
    }
    GROUP BY ?from ?to
  }
  BIND(IF(EXISTS { ?to riff:hasStylisticOrigin ?from }, "direct",
       IF(EXISTS { ?to riff:hasStylisticOrigin/riff:hasStylisticOrigin ?from }, "two-step",
       IF(EXISTS { ?from riff:hasStylisticOrigin|(riff:hasStylisticOrigin/riff:hasStylisticOrigin) ?to }, "reversed",
       "hidden"))) AS ?status)
  BIND(IRI(CONCAT("https://w3id.org/riff/route/", STRAFTER(STR(?from), "genre/"), "--", STRAFTER(STR(?to), "genre/"))) AS ?route)
}

Result: an RDF graph, saved as results/09_build_genre_routes.ttl.

14. Where artist pairs come from

How many artist pairs come only from Wikidata, only from Wikipedia text, or from both?

SELECT ?source (COUNT(*) AS ?artistPairs)
WHERE {
  ?influence a riff:Influence .
  BIND(EXISTS { ?claim riff:describes ?influence ; riff:hasEvidenceKind riff:Unsourced } AS ?inWikidata)
  BIND(EXISTS { ?claim riff:describes ?influence ; riff:hasEvidenceKind ?kind . FILTER(?kind IN (riff:Declared, riff:Attributed)) } AS ?inText)
  BIND(IF(?inWikidata && ?inText, "both", IF(?inWikidata, "Wikidata only", IF(?inText, "Wikipedia text only", "recordings only"))) AS ?source)
}
GROUP BY ?source
ORDER BY DESC(?artistPairs)

    SELECT ?genre (COUNT(?member) AS ?artistsInGenre)

17. Where instruments come from

Is an instrument an artist plays also played by an artist who influenced them, by an artist

SELECT ?source (COUNT(*) AS ?artistInstruments)
WHERE {
  ?artist riff:playsInstrument ?instrument ;
          riff:hasPrimaryGenre ?genre .
  FILTER(?instrument != instrument:voice)
  BIND(EXISTS { ?artist mm:isInfluencedBy ?other . ?other riff:playsInstrument ?instrument } AS ?fromInfluences)
  BIND(EXISTS {
    ?genre riff:hasStylisticOrigin ?root .
    ?rootArtist riff:hasPrimaryGenre ?root ;
                riff:playsInstrument ?instrument .
  } AS ?fromRoots)
  BIND(IF(?fromInfluences && ?fromRoots, "both",
       IF(?fromInfluences, "influences only",
       IF(?fromRoots, "genre roots only", "neither"))) AS ?source)
}
GROUP BY ?source
ORDER BY DESC(?artistInstruments)

17b. Where each instrument comes from

The same as query 17, but for each instrument.

       (SUM(IF(?fromInfluences && ?fromRoots, 1, 0)) AS ?both)
       (SUM(IF(?fromInfluences && !?fromRoots, 1, 0)) AS ?influencesOnly)
       (SUM(IF(!?fromInfluences && ?fromRoots, 1, 0)) AS ?rootsOnly)
       (SUM(IF(!?fromInfluences && !?fromRoots, 1, 0)) AS ?neither)
       (COUNT(*) AS ?artists)
WHERE {
  ?artist riff:playsInstrument ?instrument ;
          riff:hasPrimaryGenre ?genre .

    ?genre riff:hasStylisticOrigin ?root .
    ?rootArtist riff:hasPrimaryGenre ?root ;
                riff:playsInstrument ?instrument .

}
